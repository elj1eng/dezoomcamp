## Question 4
Answer: East Harlem North
```sql
WITH short AS (
    SELECT pickup_zone, revenue_monthly_total_amount 
    FROM prod.fct_monthly_zone_revenue 
    WHERE revenue_month >= '2020-01-01' 
    AND revenue_month <'2021-01-01' 
    AND service_type = 'Green'
)
SELECT pickup_zone 
FROM short 
GROUP BY pickup_zone 
ORDER BY SUM(revenue_monthly_total_amount) DESC 
LIMIT 1;
```

import duckdb

    CREATE OR REPLACE TABLE prod.fhv_trip AS 
    SELECT * FROM read_csv_auto('data/fhv_data_2019/fhv_tripdata_2019-*.csv.gz');
